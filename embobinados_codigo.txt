        'Content-Type': 'application/json',
    },
});

// Interceptor opcional para manejar errores globalmente en el frontend
api.interceptors.response.use(
    (response) => response,
    (error) => {
        // Aquí capturamos el JSON limpio que arroja tu GlobalExceptionHandler de Java
        if (error.response && error.response.data) {
            console.error("Error del backend:", error.response.data.message);
        }
        return Promise.reject(error);
    }
);

    // Eliminar un APU en cascada

import { useState, useEffect } from 'react';
import { catalogService } from '../services/catalogServices';
import type { Labor } from '../types';

export default function LaborManager() {
    const [labors, setLabors] = useState<Labor[]>([]);
    const [loading, setLoading] = useState<boolean>(true);
    const [error, setError] = useState<string | null>(null);
    const [successMessage, setSuccessMessage] = useState<string | null>(null);

    const [currentLabor, setCurrentLabor] = useState<Labor>({
        id: '',
        description: '',
        baseDailyWage: 0
    });
    const [isEditing, setIsEditing] = useState<boolean>(false);

    const fetchLabors = async () => {
        try {
            setLoading(true);
            const data = await catalogService.getLabors();
            setLabors(data);
            setError(null);
        } catch (err) {
            setError('Error al cargar los cargos desde el servidor.');
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        fetchLabors();
    }, []);

    const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
        const { name, value } = e.target;
        setCurrentLabor(prev => ({
            ...prev,
            [name]: name === 'baseDailyWage' ? parseFloat(value) || 0 : value
        }));
    };

    const openCreateModal = () => {
        setIsEditing(false);
        setCurrentLabor({ id: '', description: '', baseDailyWage: 0 });
        (document.getElementById('modal_labor') as HTMLDialogElement)?.showModal();
    };

    const openEditModal = (labor: Labor) => {
        setIsEditing(true);
        setCurrentLabor(labor);
        (document.getElementById('modal_labor') as HTMLDialogElement)?.showModal();
    };

    const handleSubmit = async (e: React.FormEvent) => {
        e.preventDefault();
        setError(null);
        setSuccessMessage(null);

        try {
            if (isEditing) {
                await catalogService.updateLabor(currentLabor.id, currentLabor);
                setSuccessMessage('Cargo actualizado exitosamente.');
            } else {
                await catalogService.createLabor(currentLabor);
                setSuccessMessage('Cargo registrado exitosamente.');
            }
            fetchLabors();
            (document.getElementById('modal_labor') as HTMLDialogElement)?.close();
        } catch (err: any) {
            setError(err.response?.data?.message || 'Error al guardar el cargo.');
        }
    };

    const handleDelete = async (id: string) => {
        if (!confirm(`¿Estás seguro de eliminar el cargo con ID: ${id}?`)) return;
        setError(null);
        setSuccessMessage(null);
        try {
            await catalogService.deleteLabor(id);
            setSuccessMessage('Cargo eliminado correctamente.');
            fetchLabors();
        } catch (err: any) {
            setError(err.response?.data?.message || 'No se puede eliminar el cargo porque está en uso.');
        }

            </div>

            {error && <div className="alert alert-error mb-4 shadow-sm"><span>{error}</span></div>}
            {successMessage && <div className="alert alert-success mb-4 shadow-sm"><span>{successMessage}</span></div>}

            <div className="overflow-x-auto">
                <table className="table table-zebra w-full">
                    <thead>
                    <tr>
                        <th>ID</th>
                        <th>Descripción del Cargo</th>
                        <th>Salario Diario Base</th>
                        <th className="text-center">Acciones</th>
                    </tr>
                    </thead>
                    <tbody>
                    {loading ? (
                        <tr><td colSpan={4} className="text-center py-6">Cargando cargos...</td></tr>
                    ) : labors.length === 0 ? (

public class StringUtilz {
    public static String limpiarParametro(String valor) {
        // Si es nulo o solo contiene espacios en blanco, retorna null.
        // De lo contrario, retorna el texto sin espacios accidentales a los lados.
        return (valor != null && !valor.isBlank()) ? valor.trim() : null;
    }

    public static String prepararFiltro(String valor, boolean exactMatch) {
        if (valor == null || valor.isBlank()) return null;
        String limpio = valor.trim();
        return exactMatch ? limpio : "%" + limpio + "%";
    }
}

    public record ErrorResponse(

            int status,

            String message
    ) {}

    public ResponseEntity<ErrorResponse> handleRuntimeException(RuntimeException ex) {

        ErrorResponse error = new ErrorResponse(
                LocalDateTime.now(),
                HttpStatus.BAD_REQUEST.value(),
                "Error en la Petición",
                ex.getMessage() // Aquí inyectará "Material no encontrado", etc.
        );

        return new ResponseEntity<>(error, HttpStatus.BAD_REQUEST);
    }

@CrossOrigin(origins = "*") // Permite que el frontend en React (puerto 5173) se comunique sin bloqueos CORS
public class ApuController {

    @PostMapping("/generar")
    public ResponseEntity<Process> generarApu(@RequestBody ApuRequestDTO dto) {

        // 1. Mapear la cabecera (Process)
        Process process = new Process();
        process.setServiceRequest(dto.getServiceRequest());
        process.setMainProcess(dto.getMainProcess());
        process.setDescription(dto.getDescription());
        process.setPower(dto.getPower());
        process.setSerial(dto.getSerial());
        process.setBrand(dto.getBrand());
        process.setLocation(dto.getLocation());
        process.setItemQuantity(dto.getItemQuantity());
        process.setPerformance(dto.getPerformance());
        process.setSocialLawsPercentage(dto.getSocialLawsPercentage());
        process.setAdminPercentage(dto.getAdminPercentage());
        process.setUtilityContingencyPercentage(dto.getUtilityContingencyPercentage());

        // Truco JPA: Para guardar relaciones, solo necesitamos instanciar la entidad con su ID
        Company company = new Company();
        company.setId(dto.getCompanyId());
        process.setCompany(company);

        // 2. Mapear la lista de Materiales
        List<ProcessMaterial> materials = dto.getMaterials().stream().map(mDto -> {
            ProcessMaterial pm = new ProcessMaterial();
            Material material = new Material();
            material.setId(mDto.getMaterialId());
            pm.setMaterial(material);
            pm.setQuantity(mDto.getQuantity());
            return pm;
        }).collect(Collectors.toList());

        // 3. Mapear la lista de Equipos
        List<ProcessEquipment> equipments = dto.getEquipments().stream().map(eDto -> {
            ProcessEquipment pe = new ProcessEquipment();
            Equipment equipment = new Equipment();
            equipment.setId(eDto.getEquipmentId());
            pe.setEquipment(equipment);
            pe.setQuantity(eDto.getQuantity());
            return pe;
        }).collect(Collectors.toList());

        // 4. Mapear la lista de Mano de Obra
        List<ProcessLabor> labors = dto.getLabors().stream().map(lDto -> {
            ProcessLabor pl = new ProcessLabor();
            Labor labor = new Labor();
            labor.setId(lDto.getLaborId());
            pl.setLabor(labor);
            pl.setDotation(lDto.getDotation());
            return pl;
        }).collect(Collectors.toList());

        // 5. Ejecutar el cálculo y guardar
        Process savedProcess = apuService.calcAndSaveApu(process, materials, equipments, labors);

        // Devolvemos el APU calculado y guardado con un código HTTP 200 (OK)
        return ResponseEntity.ok(savedProcess);

package com.embobinados.apugen_backend.service;

import com.embobinados.apugen_backend.dto.ApuResponseDTO;
import com.embobinados.apugen_backend.entity.*;
import com.embobinados.apugen_backend.entity.Process;
import com.embobinados.apugen_backend.repository.*;
import com.embobinados.apugen_backend.utilz.StringUtilz;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;

@Service
@RequiredArgsConstructor
public class ApuService {
    private final ProcessRepository processRepository;
    private final MaterialRepository materialRepository;
    private final EquipmentRepository equipmentRepository;
    private final LaborRepository laborRepository;
    private final ProcessMaterialRepository processMaterialRepository;
    private final ProcessEquipmentRepository processEquipmentRepository;
    private final ProcessLaborRepository processLaborRepository;

    @Transactional
    public Process calcAndSaveApu(
            Process process,
            List<ProcessMaterial> materials,
            List<ProcessEquipment> equipments,
            List<ProcessLabor> labors
    ) {
        if (process.getPerformance() == null || process.getPerformance().compareTo(BigDecimal.ZERO) <= 0) {
            throw new RuntimeException("El rendimiento del APU debe ser mayor a cero para poder calcular los costos.");
        }

        // CALCULO DE MATERIALES
        BigDecimal totalMateriales = BigDecimal.ZERO;
        for (ProcessMaterial pm : materials) {
            // Ir al catalogo, extraer datos y hacer el snapshot
            Material catMat = materialRepository.findById(pm.getMaterial().getId())
                    .orElseThrow(() -> new RuntimeException("Material no encontrado"));

            pm.setDescriptionSnapshot(catMat.getDescription());
            pm.setUnitSnapshot(catMat.getUnit());
            pm.setCostApplied(catMat.getBaseCost());

            // monto = costo * cantidad
            BigDecimal amount = pm.getCostApplied().multiply(pm.getQuantity()).setScale(2, RoundingMode.HALF_UP);
            pm.setAmount(amount);
            pm.setProcess(process);

            totalMateriales = totalMateriales.add(amount);
        }

        //costo_unitario_materiales = total_materiales * cantidad_item
        BigDecimal materialUnitCost = totalMateriales.multiply(process.getItemQuantity()).setScale(2, RoundingMode.HALF_UP);
        process.setMaterialUnitCost(materialUnitCost);

        // CALCULO DE EQUIPOS

        BigDecimal totalEquipos = BigDecimal.ZERO;
        for (ProcessEquipment pe : equipments) {
            // Ir al inventario, extraer datos y hacer el snapshot
            Equipment catEq = equipmentRepository.findById(pe.getEquipment().getId())
                    .orElseThrow(() -> new RuntimeException("Equipo no encontrado"));

            pe.setDescriptionSnapshot(catEq.getDescription());
            pe.setValueApplied(catEq.getBaseValue());
            pe.setDepreciationFactorApplied(catEq.getDepreciationFactor()); // Se mantiene con 5 decimales

            // costo_uso_diario = valor * factor_depreciacion
            BigDecimal dailyUseCost = pe.getValueApplied().multiply(pe.getDepreciationFactorApplied().setScale(6, RoundingMode.HALF_UP));
            pe.setDailyUseCost(dailyUseCost);
            pe.setProcess(process);

            totalEquipos = totalEquipos.add(dailyUseCost);
        }

        // costo_unitario_equipos = total_equipos / rendimiento
        BigDecimal equipmentUnitCost = totalEquipos.divide(process.getPerformance(), 2, RoundingMode.HALF_UP);
        process.setEquipmentUnitCost(equipmentUnitCost);

        // CALCULO DE MANO DE OBRA
        BigDecimal totalEnMano = BigDecimal.ZERO;
        for (ProcessLabor pl : labors) {
            Labor catLab = laborRepository.findById(pl.getLabor().getId())
                    .orElseThrow(() -> new RuntimeException("Mano de Obra no encontrada"));

            pl.setDescriptionSnapshot(catLab.getDescription());
            pl.setDailyWageApplied(catLab.getBaseDailyWage());

            // costo_diario = salario * dotacion
            BigDecimal dailyCost = pl.getDailyWageApplied().multiply(pl.getDotation()).setScale(2, RoundingMode.HALF_UP);
            pl.setDailyCost(dailyCost);
            pl.setProcess(process);

            totalEnMano = totalEnMano.add(dailyCost);
        }

        // costo_por_leyes_sociales = total_en_mano * socialLawsPercentage
        BigDecimal costoLeyesSociales = totalEnMano.multiply(process.getSocialLawsPercentage());
        // costo_diario_personal = total_en_mano + costo_por_leyes_sociales
        BigDecimal costoDiarioPersonal = totalEnMano.add(costoLeyesSociales);

        // costo_unitario_mano_obra = costo_diario_personal / rendimiento
        BigDecimal laborUnitCost = costoDiarioPersonal.divide(process.getPerformance(), 2, RoundingMode.HALF_UP);
        process.setLaborUnitCost(laborUnitCost );

        // TOTALES GENERALES DEL PROCESO
        // costo_unitario_directo = sumatoria de las 3 listas
        BigDecimal directUnitCost = materialUnitCost.add(equipmentUnitCost).add(laborUnitCost);
        process.setDirectUnitCost(directUnitCost);

        // admin_cost_value = directUnitCost * adminPercentage
        BigDecimal adminCostValue = directUnitCost.multiply(process.getAdminPercentage()).setScale(2, RoundingMode.HALF_UP);
        process.setAdminCostValue(adminCostValue);

        // totalUnitCost = directUnitCost + adminCostValue
        BigDecimal totalUnitCost = directUnitCost.add(adminCostValue);
        process.setTotalUnitCost(totalUnitCost);

        // utilityCostValue = totalUnitCost * utilityContingencyPercentage
        BigDecimal utilityCostValue = totalUnitCost.multiply(process.getUtilityContingencyPercentage()).setScale(2, RoundingMode.HALF_UP);
        process.setUtilityCostValue(utilityCostValue);

        // calculatedUnitPrice = totalUnitCost + utilityCostValue
        BigDecimal calculatedUnitPrice = totalUnitCost.add(utilityCostValue);
        process.setCalculatedUnitPrice(calculatedUnitPrice);

        // precio_unitario_propuesto (entero)
        process.setProposedUnitPrice(calculatedUnitPrice.setScale(0, RoundingMode.HALF_UP).intValue());

        // GUARDAR EN LA BASE DE DATOS
        // Inyectamos las listas dentro del proceso para completar la relación bidireccional
        process.setMaterials(materials);
        process.setEquipments(equipments);
        process.setLabors(labors);

        // Al guardar la cabecera, Hibernate detecta las listas internas y guarda todo en cascada
        return processRepository.save(process);
    }

    @Transactional
    public void deleteApu(String serviceRequest) {
        if (!processRepository.existsById(serviceRequest)) {
            throw new RuntimeException("No se puede eliminar: No se encontró ningún APU con la solicitud " + serviceRequest);
        }

        // Al eliminar el padre, la base de datos elimina los detalles automáticamente
        processRepository.deleteById(serviceRequest);
    }

    public ApuResponseDTO getApuByServiceRequest(String serviceRequest) {
        // 1. Buscar la cabecera principal
        Process process = processRepository.findById(serviceRequest)
                .orElseThrow(() -> new RuntimeException("No se encontró ningún APU con la solicitud: " + serviceRequest));

        // 2. Buscar las 3 listas usando los métodos que acabamos de crear
        List<ProcessMaterial> materials = processMaterialRepository.findByProcessServiceRequest(serviceRequest);
        List<ProcessEquipment> equipments = processEquipmentRepository.findByProcessServiceRequest(serviceRequest);
        List<ProcessLabor> labors = processLaborRepository.findByProcessServiceRequest(serviceRequest);

        // 3. Empaquetar la informacion en el DTO y devolverlo
        return new ApuResponseDTO(process, materials, equipments, labors);
    }

    public List<Process> getAllApus() {
        // findAll() busca todas las cabeceras de la tabla 'processes'
        return processRepository.findAll();
    }

    public List<Process> searchApus(String companyId, String companyName, String mainProcess, String description, String power, String serial, String brand, String location, boolean exactMatch) {

        // Limpiamos strings vacíos para evitar falsos positivos en el IS NULL
        companyId = StringUtilz.prepararFiltro(companyId, exactMatch);
        companyName = StringUtilz.prepararFiltro(companyName, exactMatch);
        mainProcess = StringUtilz.prepararFiltro(mainProcess, exactMatch);
        description = StringUtilz.prepararFiltro(description, exactMatch);
        power = StringUtilz.prepararFiltro(power, exactMatch);
        serial = StringUtilz.prepararFiltro(serial, exactMatch);
        brand = StringUtilz.prepararFiltro(brand, exactMatch);
        location = StringUtilz.prepararFiltro(location, exactMatch);

        // Si no consigue nada, Spring Data JPA devuelve [] automáticamente.
        return processRepository.searchApus(companyId, companyName, mainProcess, description, power, serial, brand, location);
    }

package com.embobinados.apugen_backend.repository;

import com.embobinados.apugen_backend.entity.Process;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.util.List;

@Repository
public interface ProcessRepository extends JpaRepository<Process, String> {

    @Query("""
    SELECT p FROM Process p 
    WHERE (:companyId IS NULL OR normalizar(cast(p.company.id as String)) LIKE normalizar(cast(:companyId as String)))
      AND (:companyName IS NULL OR normalizar(p.company.name) LIKE normalizar(cast(:companyName as String)))
      AND (:mainProcess IS NULL OR normalizar(p.mainProcess) LIKE normalizar(cast(:mainProcess as String)))
      AND (:description IS NULL OR normalizar(p.description) LIKE normalizar(cast(:description as String)))
      AND (:power IS NULL OR normalizar(p.power) LIKE normalizar(cast(:power as String)))
      AND (:serial IS NULL OR normalizar(p.serial) LIKE normalizar(cast(:serial as String)))
      AND (:brand IS NULL OR normalizar(p.brand) LIKE normalizar(cast(:brand as String)))
      AND (:location IS NULL OR normalizar(p.location) LIKE normalizar(cast(:location as String)))
""")
    List<Process> searchApus(
            @Param("companyId") String companyId,
            @Param("companyName") String companyName,
            @Param("mainProcess") String mainProcess,
            @Param("description") String description,
            @Param("power") String power,
            @Param("serial") String serial,
            @Param("brand") String brand,
            @Param("location") String location
    );
}
