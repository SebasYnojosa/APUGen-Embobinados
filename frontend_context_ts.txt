import axios from 'axios';

// Creamos una instancia de Axios con la URL base de nuestro Spring Boot
const api = axios.create({
    baseURL: 'http://localhost:8080/api',
    headers: {
        'Content-Type': 'application/json',
    },
});

// Interceptor opcional para manejar errores globalmente en el frontend
api.interceptors.response.use(
    (response) => response,
    (error) => {
        // Aquí capturamos el JSON limpio que arroja tu GlobalExceptionHandler de Java
        if (error.response && error.response.data) {
            console.error("Error del backend:", error.response.data.message);
        }
        return Promise.reject(error);
    }
);

export default api;

import api from './api';
import type { Process, ApuResponseDTO } from '../types';

export const apuService = {
    // Obtener todos los APUs o filtrar por empresa
    getAllApus: async (companyId?: string): Promise<Process[]> => {
        const params = companyId ? { companyId } : {};
        const response = await api.get<Process[]>('/apus', { params });
        return response.data;
    },

    // Buscador avanzado con múltiples parámetros dinámicos
    searchApus: async (filters: Record<string, string | number | boolean>): Promise<Process[]> => {
        const response = await api.get<Process[]>('/apus/buscar', { params: filters });
        return response.data;
    },

    // Obtener un APU específico con todas sus listas de detalles
    getApuById: async (serviceRequest: string): Promise<ApuResponseDTO> => {
        const response = await api.get<ApuResponseDTO>(`/apus/${serviceRequest}`);
        return response.data;
    },

    // Eliminar un APU en cascada
    deleteApu: async (serviceRequest: string): Promise<void> => {
        await api.delete(`/apus/${serviceRequest}`);
    }
};

// ==========================================
// CATÁLOGOS MAESTROS
// ==========================================
export interface Company {
    id: string;
    name: string;
}

export interface Material {
    id: string;
    description: string;
    unit: string;
    baseCost: number;
}

export interface Equipment {
    id: string;
    description: string;
    baseValue: number;
    depreciationFactor: number;
}

export interface Labor {
    id: string;
    description: string;
    baseDailyWage: number;
}

// ==========================================
// DETALLES DEL PROCESO (SNAPSHOTS)
// ==========================================
export interface ProcessMaterial {
    id: string;
    material: Material;
    descriptionSnapshot: string;
    unitSnapshot: string;
    costApplied: number;
    quantity: number;
    amount: number;
}

export interface ProcessEquipment {
    id: string;
    equipment: Equipment;
    descriptionSnapshot: string;
    valueApplied: number;
    depreciationFactorApplied: number;
    quantity: number;
    dailyUseCost: number;
}

export interface ProcessLabor {
    id: string;
    labor: Labor;
    descriptionSnapshot: string;
    dailyWageApplied: number;
    dotation: number;
    dailyCost: number;
}

// ==========================================
// CABECERA DEL APU (PROCESO)
// ==========================================
export interface Process {
    serviceRequest: string;
    company: Company;
    mainProcess: string;
    description: string;
    power?: string;    // Opcional
    serial?: string;   // Opcional
    brand?: string;    // Opcional
    location?: string;
    
    // Variables de Cálculo
    itemQuantity: number;
    performance: number;
    socialLawsPercentage: number;
    adminPercentage: number;
    utilityContingencyPercentage: number;
    
    // Totales
    materialUnitCost: number;
    equipmentUnitCost: number;
    laborUnitCost: number;
    directUnitCost: number;
    adminCostValue: number;
    totalUnitCost: number;
    utilityCostValue: number;
    calculatedUnitPrice: number;
    proposedUnitPrice: number;
}

// ==========================================
// DTO DE RESPUESTA COMPLETA (Para el endpoint GET)
// ==========================================
export interface ApuResponseDTO {
    process: Process;
    materials: ProcessMaterial[];
    equipments: ProcessEquipment[];
    labors: ProcessLabor[];
}
